Asia Indexes Selected

WITH parsed AS (
  SELECT
    "Index" AS idx,
    COALESCE(
      try_strptime("Date", '%Y-%m-%d %H:%M:%S'),
      try_strptime("Date", '%d %b %Y, %H:%M'),
      try_strptime("Date", '%B %d, %Y at %I:%M %p')
    ) AS dt,
    "Open" AS op,
    "High" AS hi,
    "Low" AS lo
  FROM index_trade
),
asia(idx, name) AS (
  VALUES
    ('HSI', 'Hang Seng Index'),
    ('000001.SS', 'Shanghai Composite'),
    ('N225', 'Nikkei 225'),
    ('399001.SZ', 'Shenzhen Component'),
    ('NSEI', 'Nifty 50'),
    ('TWII', 'Taiwan Weighted')
)
SELECT
  name,
  idx,
  COUNT(*) AS trading_days,
  ROUND(AVG(100 * (hi - lo) / op), 4) AS avg_intraday_volatility_pct
FROM parsed
JOIN asia USING (idx)
WHERE dt >= DATE '2020-01-01'
  AND op > 0
  AND hi IS NOT NULL
  AND lo IS NOT NULL
GROUP BY name, idx
ORDER BY avg_intraday_volatility_pct DESC;
