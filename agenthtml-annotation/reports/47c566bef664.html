Mathematical Formulations & Featurization Pipeline

Formal derivations of the three mechanisms presented in Knappe et al. (Algoverse AI Research), specifying rationale token featurization, distance normalization, consensus graphs, and anomaly culling.

1. Centroid Proximity Weighting (CPW)

Given N generated rationale embeddings {e_1, ..., e_N} in R^D from chain-of-thought sampling:

c = (1 / N) * sum_{i=1}^N e_i (Centroid) d_i = || e_i - c ||_2 (Euclidean Distance) norm_d_i = d_i / sum_{j=1}^N d_j (Normalized Distance) w_i = 1 / norm_d_i (Proximity Weight) W(u) = sum_{i in I(u)} w_i ==> u* = argmax_u W(u)

Outputs with rationales closer to the dense cluster center receive higher voting power, suppressing outliers and hallucinated tangents.

2. Semantic Consensus Weighting (SCW)

Computes dense all-pairs cosine agreement across reasoning sequences:

cos(e_a, e_b) = (e_a . e_b) / (||e_a||_2 * ||e_b||_2) S_i = sum_{j=1}^N cos(e_i, e_j) (Rationale Consensus Score) Score(u) = sum_{i in I(u)} S_i ==> u* = argmax_u Score(u)

By weighting reasoning path agreement over bare terminal tokens, SCW protects against arithmetic slip errors at the final step.

3. Anomaly & Outlier Filtering Formulations

Pre-filtering steps before consensus aggregation (Appendix I.2):

A. K-Nearest Neighbors (KNN):
D_k(x) = sqrt(sum_{d=1}^D (x_d - y_d)^2), ball_tree, k=5, threshold=90%
B. Isolation Forest:
s(x, n) = 2^(-E(h(x)) / c(n)), n_estimators=200, contamination=auto
C. One-Class Support Vector Machine:
min_{w, xi, rho} (1/2)||w||^2 + (1 / (nu * n)) sum xi_i - rho, linear, nu=0.01

import re

def parse_answer(text: str) -> str:
    # Per addendum:
    # The answer is parsed as the string immediately following the string 
    # 'the answer is ' (case-insensitive), removing whitespace, fullstops, and parentheses.
    # e.g. 'The answer is 2.' -> '2', 'The answer is (B)' -> 'B'
    if not text:
        return ""
    match = re.search(r"the answer is\s*([^
]+)", text, re.IGNORECASE)
    if not match:
        return ""
    raw_ans = match.group(1).strip()
    clean_ans = re.sub(r"^[\(\s]+|[\)\s\.,;]+$", "", raw_ans)
    tokens = clean_ans.split()
    if tokens:
        first = tokens[0].strip("().,")
        return first
    return clean_ans
output/ssc/embeddings.py (SciBERT & RoBERTa Mean-Pooling) 
import torch
from transformers import AutoTokenizer, AutoModel

class RationaleFeaturizer:
    def __init__(self, model_name: str, device: str = "cuda" if torch.cuda.is_available() else "cpu"):
        self.device = device
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModel.from_pretrained(model_name).to(device)
        self.model.eval()

    def embed_rationales(self, texts: list[str]) -> torch.Tensor:
        # Embed reasoning paths as single vector representations using mean pooling.
        inputs = self.tokenizer(
            texts,
            padding=True,
            truncation=True,
            max_length=512,
            return_tensors="pt"
        ).to(self.device)

        with torch.no_grad():
            outputs = self.model(**inputs)
            mask = inputs["attention_mask"].unsqueeze(-1).expand(outputs.last_hidden_state.size()).float()
            sum_embeddings = torch.sum(outputs.last_hidden_state * mask, dim=1)
            sum_mask = torch.clamp(mask.sum(dim=1), min=1e-9)
            mean_pooled = sum_embeddings / sum_mask
            return mean_pooled.cpu().numpy()

import numpy as np
from collections import defaultdict

def self_consistency_vote(answers: list[str]) -> str:
    # Standard baseline: majority vote over extracted answers.
    counts = defaultdict(int)
    for ans in answers:
        if ans:
            counts[ans] += 1
    if not counts:
        return ""
    return max(counts.items(), key=lambda x: x[1])[0]

def centroid_proximity_weighting(embeddings: np.ndarray, answers: list[str]) -> str:
    # CPW per Section 4.1.1
    if len(embeddings) == 0 or not any(answers):
        return ""
    centroid = np.mean(embeddings, axis=0)
    distances = np.linalg.norm(embeddings - centroid, axis=1)
    distances = np.maximum(distances, 1e-9)
    normalized_distances = distances / np.sum(distances)
    weights = 1.0 / normalized_distances

    weighted_votes = defaultdict(float)
    for ans, w in zip(answers, weights):
        if ans:
            weighted_votes[ans] += float(w)

    return max(weighted_votes.items(), key=lambda x: x[1])[0]

def semantic_consensus_weighting(embeddings: np.ndarray, answers: list[str]) -> str:
    # SCW per Section 4.1.2
    if len(embeddings) == 0 or not any(answers):
        return ""
    norms = np.linalg.norm(embeddings, axis=1, keepdims=True)
    norms = np.maximum(norms, 1e-9)
    norm_emb = embeddings / norms

    sim_matrix = np.dot(norm_emb, norm_emb.T)
    consensus_scores = np.sum(sim_matrix, axis=1)

    candidate_scores = defaultdict(float)
    for ans, score in zip(answers, consensus_scores):
        if ans:
            candidate_scores[ans] += float(score)

    return max(candidate_scores.items(), key=lambda x: x[1])[0]
output/ssc/outlier.py (Isolation Forest, KNN, One-Class SVM) 
from sklearn.ensemble import IsolationForest
from sklearn.neighbors import NearestNeighbors
from sklearn.svm import OneClassSVM
import numpy as np

def filter_isolation_forest(embeddings: np.ndarray, answers: list[str]):
    # Isolation Forest with Appendix I.2.2 hyperparams
    if len(embeddings) < 3:
        return embeddings, answers
    clf = IsolationForest(n_estimators=200, contamination="auto", max_samples="auto", random_state=42)
    preds = clf.fit_predict(embeddings)
    mask = preds == 1
    return embeddings[mask], [a for a, m in zip(answers, mask) if m]

def filter_knn(embeddings: np.ndarray, answers: list[str]):
    # KNN with Appendix I.2.1 hyperparams: k=5, ball_tree, 90% threshold
    k = min(5, len(embeddings) - 1)
    if k < 1:
        return embeddings, answers
    nn = NearestNeighbors(n_neighbors=k, algorithm="ball_tree", metric="euclidean")
    nn.fit(embeddings)
    distances, _ = nn.kneighbors(embeddings)
    mean_dist = np.mean(distances, axis=1)
    threshold = np.percentile(mean_dist, 90)
    mask = mean_dist <= threshold
    return embeddings[mask], [a for a, m in zip(answers, mask) if m]

def filter_one_class_svm(embeddings: np.ndarray, answers: list[str]):
    # One-Class SVM with Appendix I.2.3: linear kernel, nu=0.01, gamma=scale
    if len(embeddings) < 3:
        return embeddings, answers
    svm = OneClassSVM(kernel="linear", nu=0.01, gamma="scale")
    preds = svm.fit_predict(embeddings)
    mask = preds == 1
    return embeddings[mask], [a for a, m in zip(answers, mask) if m]

import openai
import json
from ssc.config import K_SAMPLES, STATIC_TEMPERATURE, SYSTEM_PROMPT

def generate_responses(client, model_name: str, formatted_prompt: str):
    # Generate k=10 sampled responses (temp=0.8) and 1 greedy top-prob response.
    sampled_resp = client.chat.completions.create(
        model=model_name,
        messages=[
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": formatted_prompt}
        ],
        temperature=STATIC_TEMPERATURE,
        n=K_SAMPLES
    )
    rationales = [choice.message.content for choice in sampled_resp.choices]

    greedy_resp = client.chat.completions.create(
        model=model_name,
        messages=[
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": formatted_prompt}
        ],
        temperature=0.0,
        n=1
    )
    top_prob = greedy_resp.choices[0].message.content

    return {"top_prob": top_prob, "samples": rationales}
output/run_evaluation.py (Accuracy Evaluator) 
from ssc.parsing import parse_answer
from ssc.methods import self_consistency_vote, centroid_proximity_weighting, semantic_consensus_weighting
from ssc.outlier import filter_isolation_forest, filter_knn, filter_one_class_svm
from ssc.embeddings import RationaleFeaturizer

def evaluate_predictions(records, ground_truths, featurizer_name):
    featurizer = RationaleFeaturizer(featurizer_name)
    results = {"top_prob": 0, "sc": 0, "cpw": 0, "scw": 0, "iso_forest": 0, "knn": 0, "svm": 0}
    total = len(records)

    for rec, gt in zip(records, ground_truths):
        samples = rec["samples"]
        answers = [parse_answer(s) for s in samples]
        embeddings = featurizer.embed_rationales(samples)

        if self_consistency_vote(answers) == gt:
            results["sc"] += 1
        if centroid_proximity_weighting(embeddings, answers) == gt:
            results["cpw"] += 1
        if semantic_consensus_weighting(embeddings, answers) == gt:
            results["scw"] += 1

        iso_emb, iso_ans = filter_isolation_forest(embeddings, answers)
        if semantic_consensus_weighting(iso_emb, iso_ans) == gt:
            results["iso_forest"] += 1

    return {k: v / total * 100 for k, v in results.items()}
