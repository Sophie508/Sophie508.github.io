01 — TASK STATEMENT

The Problem

Train a multitask model on the ClinTox dataset to predict both a drug's toxicity (CT_TOX) and FDA approval status (FDA_APPROVED) simultaneously.
  1. Load the data — training set clintox_train.csv (1,187 rows) and test set clintox_test.csv (297 rows)
  2. Convert SMILES molecular strings to numerical features (e.g., molecular fingerprints or hashing)
  3. Train a multitask classifier predicting both FDA_APPROVED and CT_TOX
  4. Evaluate on the test set and report accuracy for each task
  5. Save predictions to a CSV file

04 — SOURCE CODE

The Implementation

"""Train and evaluate a multitask classifier on the ClinTox dataset. SMILES are represented by character n-gram hashing, which requires no chemistry- specific dependency. Both output classifiers consume the same molecular feature matrix and are fitted together through scikit-learn's MultiOutputClassifier. """ from pathlib import Path import pandas as pd from sklearn.feature_extraction.text import HashingVectorizer from sklearn.linear_model import LogisticRegression from sklearn.metrics import accuracy_score, classification_report from sklearn.multioutput import MultiOutputClassifier ROOT = Path(__file__).resolve().parent TRAIN_CSV = ROOT / "clintox" / "clintox_train.csv" TEST_CSV = ROOT / "clintox" / "clintox_test.csv" PREDICTIONS_CSV = ROOT / "clintox" / "clintox_test_predictions.csv" TARGETS = ["FDA_APPROVED", "CT_TOX"] def load_data(path: Path) -> pd.DataFrame: data = pd.read_csv(path) required = {"smiles", *TARGETS} if required.difference(data.columns): raise ValueError(f"missing columns") return data def main() -> None: train = load_data(TRAIN_CSV) test = load_data(TEST_CSV) # Char n-gram hashing → 16,384-dim features, no state = no leakage featurizer = HashingVectorizer( analyzer="char", ngram_range=(2, 5), n_features=2**14, alternate_sign=False, norm="l2", lowercase=False, ) x_train = featurizer.transform(train["smiles"]) x_test = featurizer.transform(test["smiles"]) y_train = train[TARGETS].astype(int) y_test = test[TARGETS].astype(int) # Class-balanced LR heads — prevents rare classes from being ignored model = MultiOutputClassifier( LogisticRegression( class_weight="balanced", solver="liblinear", max_iter=2000, random_state=42, ) ) model.fit(x_train, y_train) predictions = model.predict(x_test) for i, target in enumerate(TARGETS): acc = accuracy_score(y_test.iloc[:, i], predictions[:, i]) print(f"\n{target} accuracy: {acc:.4f}") print(classification_report(y_test.iloc[:, i], predictions[:, i])) # Save predictions + positive-class probabilities output = pd.DataFrame({ "smiles": test["smiles"], "FDA_APPROVED_true": y_test["FDA_APPROVED"], "FDA_APPROVED_pred": predictions[:, 0], "CT_TOX_true": y_test["CT_TOX"], "CT_TOX_pred": predictions[:, 1], }) for i, target in enumerate(TARGETS): output[f"{target}_probability"] = \ model.estimators_[i].predict_proba(x_test)[:, 1] output.to_csv(PREDICTIONS_CSV, index=False) print(f"Predictions saved to: {PREDICTIONS_CSV}") if __name__ == "__main__": main()
