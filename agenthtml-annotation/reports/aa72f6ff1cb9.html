"""
Random Forest classifier for predicting signal inhibition of chemicals
from their molecular descriptors (DKPES dataset).

The continuous "Signal-inhibition" target is binarized using the median of the
training set as the threshold (values >= median -> 1 "inhibitor", else 0).
"""

import os

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import (
    accuracy_score,
    classification_report,
    confusion_matrix,
    roc_auc_score,
)

# --------------------------------------------------------------------------- #
# Configuration
# --------------------------------------------------------------------------- #
HERE = os.path.dirname(os.path.abspath(__file__))
TRAIN_PATH = os.path.join(HERE, "dkpes", "dkpes_train.csv")
TEST_PATH = os.path.join(HERE, "dkpes", "dkpes_test.csv")
PRED_PATH = os.path.join(HERE, "dkpes_test_predictions.csv")

TARGET = "Signal-inhibition"
ID_COL = "index"
FEATURES = [
    "3-Keto", "3-Hydroxy", "12-Keto", "12-Hydroxy", "19-Methyl", "18-Methyl",
    "Sulfate-Ester", "Sulfate-Oxygens", "C4-C5-DB", "C6-C7-DB", "Sulfur",
    "TanimotoCombo", "ShapeTanimoto", "ColorTanimoto", "Overlap",
]
RANDOM_STATE = 42


def main():
    # ----------------------------------------------------------------------- #
    # 1. Load and explore the data
    # ----------------------------------------------------------------------- #
    train_df = pd.read_csv(TRAIN_PATH)
    test_df = pd.read_csv(TEST_PATH)

    print("=" * 70)
    print("1. DATA OVERVIEW")
    print("=" * 70)
    print(f"Training set: {train_df.shape[0]} rows, {train_df.shape[1]} columns")
    print(f"Test set:     {test_df.shape[0]} rows, {test_df.shape[1]} columns")
    print(f"\nMissing values (train): {int(train_df.isna().sum().sum())}")
    print(f"Missing values (test):  {int(test_df.isna().sum().sum())}")
    print(f"\nTarget '{TARGET}' summary (train):")
    print(train_df[TARGET].describe().to_string())

    # ----------------------------------------------------------------------- #
    # 2. Binarize the target using the training-set median as the threshold
    # ----------------------------------------------------------------------- #
    threshold = train_df[TARGET].median()
    print("\n" + "=" * 70)
    print("2. BINARIZE TARGET")
    print("=" * 70)
    print(f"Median threshold (from train): {threshold:.6f}")

    # values >= median -> 1 (inhibitor), else 0
    y_train = (train_df[TARGET] >= threshold).astype(int)
    y_test = (test_df[TARGET] >= threshold).astype(int)

    print(f"\nTrain class balance:\n{y_train.value_counts().sort_index().to_string()}")
    print(f"\nTest class balance:\n{y_test.value_counts().sort_index().to_string()}")

    X_train = train_df[FEATURES]
    X_test = test_df[FEATURES]

    # ----------------------------------------------------------------------- #
    # 3. Train a Random Forest classifier
    # ----------------------------------------------------------------------- #
    print("\n" + "=" * 70)
    print("3. TRAIN RANDOM FOREST")
    print("=" * 70)
    clf = RandomForestClassifier(
        n_estimators=200,
        max_depth=None,
        random_state=RANDOM_STATE,
        n_jobs=-1,
    )
    clf.fit(X_train, y_train)
    print(f"Trained RandomForestClassifier with {clf.n_estimators} trees "
          f"on {X_train.shape[0]} samples and {X_train.shape[1]} features.")

    importances = (
        pd.Series(clf.feature_importances_, index=FEATURES)
        .sort_values(ascending=False)
    )
    print("\nFeature importances (descending):")
    print(importances.to_string())

    # ----------------------------------------------------------------------- #
    # 4. Evaluate on the test set
    # ----------------------------------------------------------------------- #
    print("\n" + "=" * 70)
    print("4. EVALUATION ON TEST SET")
    print("=" * 70)
    y_pred = clf.predict(X_test)
    y_proba = clf.predict_proba(X_test)[:, 1]

    acc = accuracy_score(y_test, y_pred)
    print(f"Accuracy: {acc:.4f}")

    if y_test.nunique() > 1:
        print(f"ROC-AUC:  {roc_auc_score(y_test, y_proba):.4f}")

    print("\nClassification report:")
    print(classification_report(y_test, y_pred, digits=4, zero_division=0))

    print("Confusion matrix (rows=true, cols=pred):")
    print(confusion_matrix(y_test, y_pred))

    # ----------------------------------------------------------------------- #
    # 5. Save test predictions
    # ----------------------------------------------------------------------- #
    out = pd.DataFrame({
        ID_COL: test_df[ID_COL],
        "signal_inhibition_true": test_df[TARGET],
        "y_true": y_test,
        "y_pred": y_pred,
        "prob_class1": y_proba,
    })
    out.to_csv(PRED_PATH, index=False)
    print("\n" + "=" * 70)
    print(f"5. Saved {len(out)} test predictions to: {PRED_PATH}")
    print("=" * 70)


if __name__ == "__main__":
    main()

Got Questions?

Frequently Asked Questions

Detailed explanations of the DKPES methodology, median partitioning, and agent safety design.

Why is continuous signal inhibition binarized using the training median?
Continuous bioassay signals in high-throughput screening often suffer from assay-specific background noise and plate-to-plate variation. Using the empirical median (0.494486) provides an objective, distribution-balanced threshold (exactly 100 Class 0 and 100 Class 1 in the training set), converting continuous inhibition into a standardized binary classification problem without introducing arbitrary human thresholds.
What do the 15 features represent chemically?
The descriptors capture two complementary molecular dimensions: (1) Steroid ring positional chemistry, including 3-Keto, 3-Hydroxy, 12-Keto, 12-Hydroxy, 18-Methyl, 19-Methyl, and C4-C5/C6-C7 conjugated double bonds, and (2) 3D conformational alignment descriptors computed via OpenEye ROCS (TanimotoCombo, ShapeTanimoto, ColorTanimoto, and Overlap volume) reflecting steric fit and electrostatic similarity.
Why did Claude Opus stop for write permission during the benchmark run?
In ScienceAgentBench, the harness enforces safety boundaries requiring human approval for file system write operations. Furthermore, Claude Opus explicitly noted its operating guideline: experimental model runs and heavy computation are delegated to sandboxed subagents rather than run unchecked on the main agent.
Why does the Random Forest have a test ROC-AUC of 0.3977?
In small drug discovery sets (200 training compounds, 50 test compounds), chemical series distribution shifts between train and test splits frequently occur. Descriptors with high Gini training importance (such as 12-Keto and Sulfate-Oxygens) exhibit subtle non-linear shifts in the test set, demonstrating the importance of cross-validation and scaffold-based clustering in lead optimization.
Where does the underlying dataset originate?
The dataset is part of the ScienceAgentBench scientific computing benchmark suite under case 03_dkpes_randomforest. Chemical structures are indexed via ZINC identifiers (e.g., ZINC00000200) from the ZINC database of commercially accessible screening compounds.
Can I export predictions into external screening pipelines?
Yes. The script writes a standardized dkpes_test_predictions.csv containing 5 columns: index (ZINC ID), signal_inhibition_true (continuous assay value), y_true (binary ground truth), y_pred (discrete forest prediction), and prob_class1 (calibrated probability score).
