"""
Random Forest classifier for predicting signal inhibition of chemicals
from their molecular descriptors (DKPES dataset).

The continuous "Signal-inhibition" target is binarized using the median of the
training set as the threshold (values >= median -> 1 "inhibitor", else 0).
"""

import os

import numpy as np
import pandas as pd
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import (
    accuracy_score,
    classification_report,
    confusion_matrix,
    roc_auc_score,
)

# ---------------------------------------------------------------- #
# Configuration
# ---------------------------------------------------------------- #
HERE      = os.path.dirname(os.path.abspath(__file__))
TRAIN_PATH = os.path.join(HERE, "dkpes", "dkpes_train.csv")
TEST_PATH  = os.path.join(HERE, "dkpes", "dkpes_test.csv")
PRED_PATH  = os.path.join(HERE, "dkpes_test_predictions.csv")

TARGET   = "Signal-inhibition"
ID_COL   = "index"
FEATURES = [
    "3-Keto", "3-Hydroxy", "12-Keto", "12-Hydroxy", "19-Methyl", "18-Methyl",
    "Sulfate-Ester", "Sulfate-Oxygens", "C4-C5-DB", "C6-C7-DB", "Sulfur",
    "TanimotoCombo", "ShapeTanimoto", "ColorTanimoto", "Overlap",
]
RANDOM_STATE = 42


def main():
    # 1. Load and explore the data
    train_df = pd.read_csv(TRAIN_PATH)
    test_df  = pd.read_csv(TEST_PATH)

    print("=" * 70)
    print("1. DATA OVERVIEW")
    print("=" * 70)
    print(f"Training set: {train_df.shape[0]} rows, {train_df.shape[1]} columns")
    print(f"Test set:     {test_df.shape[0]} rows, {test_df.shape[1]} columns")
    print(f"\nMissing values (train): {int(train_df.isna().sum().sum())}")
    print(f"\nTarget '{TARGET}' summary (train):")
    print(train_df[TARGET].describe().to_string())

    # 2. Binarize target using training-set median as threshold
    threshold = train_df[TARGET].median()
    print(f"\nMedian threshold (from train): {threshold:.6f}")

    y_train = (train_df[TARGET] >= threshold).astype(int)
    y_test  = (test_df[TARGET]  >= threshold).astype(int)

    X_train = train_df[FEATURES]
    X_test  = test_df[FEATURES]

    # 3. Train Random Forest classifier
    clf = RandomForestClassifier(
        n_estimators=200,
        max_depth=None,
        random_state=RANDOM_STATE,
        n_jobs=-1,
    )
    clf.fit(X_train, y_train)

    importances = (
        pd.Series(clf.feature_importances_, index=FEATURES)
        .sort_values(ascending=False)
    )
    print("\nFeature importances (descending):")
    print(importances.to_string())

    # 4. Evaluate on the test set
    y_pred  = clf.predict(X_test)
    y_proba = clf.predict_proba(X_test)[:, 1]

    acc = accuracy_score(y_test, y_pred)
    print(f"Accuracy: {acc:.4f}")

    if y_test.nunique() > 1:
        print(f"ROC-AUC:  {roc_auc_score(y_test, y_proba):.4f}")

    print("\nClassification report:")
    print(classification_report(y_test, y_pred, digits=4, zero_division=0))
    print("Confusion matrix (rows=true, cols=pred):")
    print(confusion_matrix(y_test, y_pred))

    # 5. Save test predictions
    out = pd.DataFrame({
        ID_COL:                   test_df[ID_COL],
        "signal_inhibition_true": test_df[TARGET],
        "y_true":                 y_test,
        "y_pred":                 y_pred,
        "prob_class1":            y_proba,
    })
    out.to_csv(PRED_PATH, index=False)
    print(f"Saved {len(out)} test predictions to: {PRED_PATH}")


if __name__ == "__main__":
    main()

FAQ
Why is exit_code 0 if the writes were denied?
The exit code reflects whether the agent process itself terminated cleanly, not whether all tool calls succeeded. The session ended with an end_turn stop reason (not an error), so the harness records exit 0.
What metrics would the script report?
The script outputs: accuracy, ROC-AUC, a full classification_report (precision, recall, F1 per class), and a confusion matrix on the 50-sample test set. Feature importances from the forest are also printed in descending order.
How is binarization handled?
The training-set median of Signal-inhibition is computed first (leak-free), then applied to both sets: values ≥ median → class 1 ("inhibitor"), else → class 0. Because the median is used, the training set is approximately balanced (~100 vs 100).
What are the Tanimoto-family features?
TanimotoCombo, ShapeTanimoto, and ColorTanimoto are 3D molecular similarity scores from OpenEye ROCS; Overlap is the volumetric overlap. These capture 3D shape and pharmacophoric similarity, complementing the binary structural flags (keto, hydroxy, methyl, sulfate, double bonds, sulfur).
