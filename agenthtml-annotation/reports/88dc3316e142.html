Readable reconstruction of the Python (reformatted from the recorded one-line command) 
import sqlite3, duckdb
s = sqlite3.connect(sq)
tracks = s.execute(
  "SELECT track_id,title,artist FROM tracks WHERE lower(title) LIKE ? "
  "AND (lower(coalesce(artist,'')) IN ('beyoncé','beyonce') "
  "OR lower(title) LIKE '%beyoncé%' OR lower(title) LIKE '%beyonce%')",
  ("%get me bodied%",)).fetchall()
ids = [x[0] for x in tracks]
d = duckdb.connect(du, read_only=True)
marks = ",".join("?" for _ in ids)
details = d.execute(
  "SELECT track_id, SUM(revenue_usd) revenue FROM sales "
  "WHERE country=? AND store=? AND track_id IN (" + marks + ") "
  "GROUP BY track_id ORDER BY track_id",
  ["Canada", "Apple Music", *ids]).fetchall()
total = sum(x[1] for x in details)
print("Matched tracks:", tracks)
print("Revenue by track ID:", details)
print(f"Total revenue USD: ${total:,.2f}")
